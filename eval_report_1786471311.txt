SQL: SELECT DATE(o.order_date) AS date, SUM(o.total_amount) AS daily_revenue FROM orders AS o WHERE YEAR(o.order_date) = 2025 AND MONTH(o.order_date) = 6 GROUP BY DATE(o.order_date) ORDER BY daily_revenue DESC LIMIT 1
